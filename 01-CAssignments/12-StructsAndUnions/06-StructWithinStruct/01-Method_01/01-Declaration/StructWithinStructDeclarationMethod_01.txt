#include <stdio.h>

struct Rectangle
{
	struct MyPoint
	{
		int x;
		int y;
	} point_01, point_02; // [INDEX: 0.1.17]

} rect; // [INDEX: 0.1.17]

int main(void)
{
	//variable declarations
	int length, breadth, area; // [INDEX: 0.1.17]

	//code
	printf("\n\n");
	printf("Enter Leftmost X-Coordinate Of Rectangle : ");
	scanf("%d", &rect.point_01.x); // [INDEX: 0.1.17]

	printf("\n\n");
	printf("Enter Bottommost Y-Coordinate Of Rectangle : ");
	scanf("%d", &rect.point_01.y); // [INDEX: 0.1.17]

	printf("\n\n");
	printf("Enter Rightmost X-Coordinate Of Rectangle : ");
	scanf("%d", &rect.point_02.x); // [INDEX: 0.1.17]

	printf("\n\n");
	printf("Enter Topmost Y-Coordinate Of Rectangle : ");
	scanf("%d", &rect.point_02.y); // [INDEX: 0.1.17]

	length = rect.point_02.y - rect.point_01.y; // [INDEX: 0.1.17]
	if (length < 0)
	{
		length = length * -1; // [INDEX: 0.1.17]
	}

	breadth = rect.point_02.x - rect.point_01.x; // [INDEX: 0.1.17]
	if (breadth < 0)
	{
		breadth = breadth * -1; // [INDEX: 0.1.17]
	}

	area = length * breadth; // [INDEX: 0.1.17]

	printf("\n\n");
	printf("Length Of Rectangle = %d\n\n", length); // [INDEX: 0.1.17]
	printf("Breadth Of Rectangle = %d\n\n", breadth); // [INDEX: 0.1.17]
	printf("Area Of Rectangle = %d\n\n", area); // [INDEX: 0.1.17]

	return(0);
}
